%% =========================================================== 1  PROBLEM
\section{Problem}

\begin{frame}{The problem}{Three-class sentiment from the text of a food review}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \footnotesize
      Shops receive thousands of reviews a day; a seller needs the \textbf{sentiment} of each text to see which products disappoint.
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item \textbf{Input:} title + body of an English review
        \item \textbf{Output:} negative, neutral or positive
        \item \textbf{Labels from stars} (weak labels, no human annotation):
      \end{itemize}
      \centering
      \begin{tabular}{lccccc}
        \toprule
        Stars & 1 & 2 & 3 & 4 & 5 \\
        \midrule
        Label & \multicolumn{2}{c}{negative} & neutral & \multicolumn{2}{c}{positive} \\
        \bottomrule
      \end{tabular}
      \vskip0.5em
      \begin{exampleblock}{Counting words is not enough}
        \scriptsize
        \emph{``Not as Advertised. Product arrived labeled as Jumbo Salted Peanuts\ldots the peanuts were actually small sized unsalted.''} A negative review with no strongly negative word.
      \end{exampleblock}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Why it is hard}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item \textbf{Skewed labels:} 78\% positive, 7.6\% neutral
          \item \textbf{Ambiguous neutral:} 3-star reviews praise and complain at once
          \item \textbf{Weak labels:} 5 stars with a complaint exist
          \item \textbf{Context:} negation (``not bad''), contrast (``good but\ldots''), sarcasm
          \item \textbf{Dirty data:} repeated reviews, HTML, very uneven length
        \end{itemize}
      \end{block}
      \begin{alertblock}{Main metric: Macro-F1}
        \footnotesize
        $\text{Macro-F1}=\frac13\sum_c F1_c$, every class counts equally. ``Always positive'' scores \textbf{78\% accuracy but only 0.29 Macro-F1}.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Data}

\begin{frame}{Data quality and cleaning}{Amazon Fine Food Reviews: 568{,}454 raw rows, 10 columns, October 1999 -- October 2012}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \footnotesize
      \textbf{Issues measured on the raw file}
      \vskip0.3em
      \setlength{\tabcolsep}{3pt}
      \begin{tabular}{lrr}
        \toprule
        Issue & Rows & Share \\
        \midrule
        Repeated review (key) & 174{,}521 & 30.70\% \\
        Repeated \texttt{Text} & 174{,}875 & 30.76\% \\
        Missing \texttt{Summary} & 27 & 0.005\% \\
        Missing \texttt{ProfileName} & 26 & 0.005\% \\
        Numerator $>$ denominator & 2 & $<$0.001\% \\
        \bottomrule
      \end{tabular}\\[1pt]
      {\scriptsize key = (\texttt{UserId}, \texttt{ProfileName}, \texttt{Time}, \texttt{Text})}
      \vskip0.6em
      \begin{alertblock}{Why repeats matter}
        \scriptsize
        No row is duplicated in \emph{all} columns: Amazon shares one review among product variants. User \texttt{A395BORC6FGVXV}'s 2-star ``Cough Medicine'' review appears under 3 product codes. A random split would put copies in both train and test: \textbf{data leakage}.
      \end{alertblock}
    \end{column}
    \begin{column}{0.50\textwidth}
      \footnotesize
      \textbf{Cleaning pipeline} (always keep the first occurrence)
      \vskip0.3em
      \centering
      \begin{tikzpicture}[
        box/.style={draw=HUSTRed, fill=HUSTRed!8, rounded corners=2pt, minimum height=0.6cm, text width=6.4cm, align=center, font=\scriptsize},
        drop/.style={font=\scriptsize\itshape, text=HUSTRed},
        arr/.style={-{Stealth[length=1.8mm]}, thick, HUSTRed},
        node distance=0.3cm]
        \node[box] (a) {\textbf{Raw file:} 568{,}454 rows};
        \node[box, below=of a] (b) {1. Drop repeated (user, name, time, text) $\to$ \textbf{393{,}933}};
        \node[box, below=of b] (c) {2. Drop numerator $>$ denominator $\to$ \textbf{393{,}931}};
        \node[box, below=of c] (d) {3. Drop texts still identical to an earlier row $\to$ \textbf{393{,}577}};
        \node[box, below=of d, fill=HUSTRed!25] (e) {4. Missing \texttt{Summary} $\to$ empty string; label from \texttt{Score}};
        \draw[arr] (a)--(b) node[midway, right=2pt, drop]{$-$174{,}521};
        \draw[arr] (b)--(c) node[midway, right=2pt, drop]{$-$2};
        \draw[arr] (c)--(d) node[midway, right=2pt, drop]{$-$354};
        \draw[arr] (d)--(e);
      \end{tikzpicture}
      \vskip0.3em
      \raggedright
      \scriptsize
      Result: \textbf{393{,}577 reviews} (69.2\% of raw). Step 3 also drops identical short reviews by different people: a deliberate trade of 0.09\% of the data to close the last leakage route.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Model input and data split}{Leakage-safe, one frozen configuration}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \small
      \begin{block}{Model input text}
        \footnotesize
        $\text{text}=\operatorname{clean}(\texttt{Summary})+\text{``. ''}+\operatorname{clean}(\texttt{Text})$\\
        clean = unescape HTML entities, delete tags and URLs, collapse spaces. \textbf{Not} lower-cased; punctuation, stop words and negations are kept (\emph{not}, \emph{no}, \emph{never} decide sentiment).
      \end{block}
      \footnotesize
      \textbf{Before:} \texttt{\ldots were bummed.<br /><br />Now, because\ldots}\\
      \textbf{After:} \texttt{\ldots were bummed. Now, because\ldots}
      \vskip0.5em
      HTML in 26.1\% of reviews, URLs in 2.1\%, encoded characters in 0.3\%. Cleaning all 393{,}577 reviews takes 19.5 s.
    \end{column}
    \begin{column}{0.47\textwidth}
      \footnotesize
      \setlength{\tabcolsep}{3pt}
      \begin{tabular}{lrrrr}
        \toprule
         & Reviews & neg & neu & pos \\
        \midrule
        Train (80\%) & 314{,}861 & 45{,}654 & 23{,}803 & 245{,}404 \\
        Validation (10\%) & 39{,}358 & \multicolumn{3}{c}{same shares} \\
        Test (10\%) & 39{,}358 & \multicolumn{3}{c}{same shares} \\
        \bottomrule
      \end{tabular}
      \vskip0.5em
      \begin{itemize}\setlength{\itemsep}{2pt}
        \item Stratified by label, seed 42 (14.5 / 7.6 / 77.9\% in every part)
        \item Train updates weights; validation picks epoch and threshold; \textbf{test is read once}
        \item Vocabulary and class weights come from \textbf{train only}
        \item Class-weighted cross-entropy, $w_c=N/(K\,N_c)$:
      \end{itemize}
      \centering
      \begin{tabular}{lrrr}
        \toprule
         & neg & neu & pos \\
        \midrule
        $w_c$ & 2.299 & 4.409 & 0.428 \\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}
\end{frame}

    \node[box, right=of att] (out) {Linear 256$\to$3\\softmax};
    \foreach \a/\b in {tok/emb,emb/lstm,lstm/att,att/out} \draw[arr] (\a) -- (\b);
  \end{tikzpicture}
  \vskip0.7em
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{BiLSTM + attention: from scratch}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item 6.25 M parameters; reads 200 words in both directions
          \item AdamW (lr $2\times10^{-3}$), clipping 1.0, early stopping; best epoch 4 (val Macro-F1 0.7504); 3.4 min on a T4
          \item Length bucketing: 1{,}230 batches per epoch; padding positions masked in attention
          \item Attention weights make the model inspectable: they usually fall on \emph{great}, \emph{not}, \emph{but}, \emph{mistake}
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{BERT-base: pre-trained, fine-tuned}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item \texttt{bert-base-uncased}: 12 layers, 109.48 M parameters, pre-trained on Wikipedia + BookCorpus; 3-class head on \texttt{[CLS]}
          \item 256 WordPiece tokens, batch 32, 2 epochs, AdamW (lr $3\times10^{-5}$), 10\% warm-up, fp16
          \item Val Macro-F1 0.7664 $\to$ \textbf{0.7864}, still rising: not saturated
          \item 64.4 min on a T4; one run, no hyperparameter search
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

%% =========================================================== 6  CONCLUSION
\section{Conclusion}

\begin{frame}{Conclusions, limitations and future work}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{Conclusions}
        \small
        \begin{itemize}\setlength{\itemsep}{4pt}
          \item \textbf{BERT-base is the better model:} Macro-F1 0.7877 vs 0.7422 (+0.0454, significant), better on every class
          \item Transfer learning beats learning from scratch here, at 19$\times$ the training time and 17.5$\times$ the parameters
          \item Neutral stays the hardest class (F1 0.49 and 0.55)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{alertblock}{Limitations}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item Weak labels: stars do not always match the text
          \item Random split: no test on future reviews
          \item Each model run once; BERT only 2 epochs (not saturated)
          \item One domain: Amazon food reviews, 1999--2012
        \end{itemize}
